**Design decisions:**
- `process.env.STATE_MODE` was a temporary shim before `.state/mode.json` existed (Phase 097). It is no longer the canonical mode source.
- Both hooks are `async` — they can call `await getCurrentMode()` without signature changes.
- Caching at module level avoids re-reading `.state/mode.json` on every command/tool invocation.
- `process.cwd()` is the project root when the plugin runs inside opencode — same assumption used by the config hook.
- The `"kernel"` mode (internal-only, not in mode.json) is not handled by these hooks — if `.state/mode.json` is absent, the hooks default to permissive (null) rather than treating "kernel" as a valid mode.
- "both" mode means the user explicitly chose permissive — no blocking occurs.

```typescript
/** Cached mode reader — wraps readModeConfig with module-level cache */

import { readModeConfig } from "./hooks/config.js";

export type ResolvedMode = "build" | "teach" | "both" | null;

let cachedMode: ResolvedMode | undefined;

/**
 * Return the active mode from .state/mode.json.
 * Reads the file once and caches the result at module scope.
 * Returns null when mode.json is absent, malformed, or contains an
 * invalid mode — callers treat null as permissive (no blocking).
 */
export async function getCurrentMode(projectDir: string): Promise<ResolvedMode> {
  if (cachedMode !== undefined) return cachedMode;
  const mode = await readModeConfig(projectDir);
  // readModeConfig returns "build" | "teach" | "both" | null
  cachedMode = mode as ResolvedMode;
  return cachedMode;
}

/**
 * Invalidate the cached mode so the next getCurrentMode() call
 * re-reads .state/mode.json. Used for hot-reload on mode change.
 */
export function resetModeCache(): void {
  cachedMode = undefined;
}
```

```typescript
export const commandExecuteBefore: NonNullable<
  Hooks["command.execute.before"]
> = async (input, output) => {
  const mode = await getCurrentMode(process.cwd());
  // Both mode and absent mode.json are permissive — allow all commands
  if (mode === "both" || mode === null) return;

  const match = input.command.match(STATE_COMMAND_RE);
  if (!match) return;

  const [, commandMode] = match;

  if (commandMode !== mode) {
    output.parts.push(
      createTextPart(
        `[state] Cross-mode command blocked: ${input.command} is not allowed in ${mode} mode.`
      )
    );
    return;
  }

  log({
    source: "@state/opencode-plugin",
    hook: "command.execute.before",
    type: "command.gated",
    command: input.command,
    mode,
    timestamp: Date.now(),
  });
};
```

```typescript
export const toolExecuteBefore: NonNullable<
  Hooks["tool.execute.before"]
> = async (input, output) => {
  const mode = await getCurrentMode(process.cwd());
  // Both mode and absent mode.json are permissive — allow all tools
  if (mode === "both" || mode === null) return;

  const tool = input.tool;

  if (mode === "build" && tool.startsWith(STATE_TEACH_PREFIX)) {
    throw new Error(
      `[state] Cross-mode tool blocked: ${tool} is forbidden in build mode`
    );
  }

  if (mode === "teach" && tool.startsWith(STATE_BUILD_PREFIX)) {
    throw new Error(
      `[state] Cross-mode tool blocked: ${tool} is forbidden in teach mode`
    );
  }

  if (isStateTool(tool) && output.args && typeof output.args === "object") {
    output.args = rewriteStatePaths(
      output.args as Record<string, unknown>
    );
  }
};
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-100-01 | Tampering | .state/mode.json content | mitigate | readModeConfig (Phase 099) validates mode ∈ {"build","teach","both"} and returns null on invalid content — hooks treat null as permissive, preventing a corrupted file from enabling cross-mode execution while also preventing denial-of-service from a missing file |
| T-100-02 | Spoofing | command.execute.before | mitigate | STATE_COMMAND_RE regex `^\/state:(build|teach):` restricts matching to exact /state: prefix — attacker cannot inject a malformed command to bypass the gate |
| T-100-03 | Spoofing | tool.execute.before | mitigate | Tool prefix checks use startsWith on known constants (STATE_BUILD_PREFIX, STATE_TEACH_PREFIX) — no regex injection surface |
| T-100-04 | Elevation of Privilege | Both hooks (null-mode path) | accept | When .state/mode.json is absent (null mode), both hooks are permissive. This is by design: mode.json is bootstrapped by `state mode init` (Phase 097/103) and its absence indicates pre-initialization state. The canonical gate (daemon HTTP middleware, layer 5) provides the authoritative block. Risk accepted as defense-in-depth layer ordering. |
| T-100-05 | Denial of Service | Cached mode reader | mitigate | getCurrentMode() caches at module scope — a single failed read does not cause repeated filesystem access on every hook invocation. resetModeCache() allows hot-reload without restart. |
