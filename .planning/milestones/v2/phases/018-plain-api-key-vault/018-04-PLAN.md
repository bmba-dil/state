    # Mode check (POSIX only)
    if [ "$(uname)" != "MINGW*" ] && [ "$(uname)" != "Darwin_skip" ]; then
      MODE="$(stat -f '%Lp' "$SMOKE_VAULT" 2>/dev/null || stat -c '%a' "$SMOKE_VAULT")"

    fi
    # Confirm vault contains the credential
    python3 -c "
    import os
    from state_core.auth.store import load_vault
    from pathlib import Path
    os.environ['STATE_AUTH_JSON'] = '$SMOKE_VAULT'
    vault = load_vault(Path('$SMOKE_VAULT'))
    assert 'openai' in vault.providers, f'openai bucket missing from {vault.providers}'
    assert len(vault.providers['openai']) == 1, f'expected 1 cred, got {len(vault.providers[\"openai\"])}'
    cred = vault.providers['openai'][0]
    assert cred.key == 'sk-smoke-test-key-not-real-XXXXX', f'key mismatch: {cred.key!r}'
    assert cred.provider_id == 'openai'
    assert cred.type == 'api_key'
    print('OK: smoke login persisted ApiKeyCredential to vault with key not in repr:', repr(cred))
