    ### Step 5: select_credential round-trip smoke test

    Run an end-to-end demo against a tmp_path-isolated vault to prove the public surface works as documented:

    ```bash
    SMOKE_DIR="$(mktemp -d)"
    SMOKE_VAULT="${SMOKE_DIR}/.state/auth.json"
    mkdir -p "$(dirname "$SMOKE_VAULT")"

    env STATE_AUTH_JSON="$SMOKE_VAULT" python3 -c "
    import asyncio
    from state_core.auth import (
        AuthVault,
        ApiKeyCredential,
        save_vault,
        load_vault,
        select_credential,
        mark_rate_limited,
        clear_rate_limited,
        iter_active_credentials,
        NoCredentialsAvailableError,
    )
    from state_core.auth.store import get_auth_json_path

    NOW = 1_770_000_000.0
    path = get_auth_json_path()

    # Pre-seed 3 ApiKeyCredentials.
    vault = AuthVault(
        providers={
            'openai': [
                ApiKeyCredential(key=f'sk-smoke-TEST-{i}', provider_id='openai')
                for i in range(3)
            ],
        },
    )
    save_vault(path, vault)

    async def run() -> None:
        # 6 rounds with bucket-advancing now: every cred should appear.
        seen = set()
        for round_i in range(6):
            idx, cred = await select_credential(
                'openai',
                vault_path=path,
                now=NOW + round_i * 60.0,
            )
            assert cred.key == f'sk-smoke-TEST-{idx}', (idx, cred.key)
            seen.add(idx)
        assert seen == {0, 1, 2}, f'missing: {set(range(3)) - seen}'

        # Cool-down skip
        mark_rate_limited('openai', 0, until=NOW + 60.0)
        idx, _ = await select_credential('openai', vault_path=path, now=NOW)
        assert idx != 0, idx
        clear_rate_limited('openai', 0)

        # All-cooled-down → NoCredentialsAvailableError
        mark_rate_limited('openai', 0, until=NOW + 60.0)
        mark_rate_limited('openai', 1, until=NOW + 60.0)
        mark_rate_limited('openai', 2, until=NOW + 60.0)
        try:
            await select_credential('openai', vault_path=path, now=NOW)
        except NoCredentialsAvailableError as e:
            assert e.reason == 'all_cooled_down'
            assert e.earliest_available_at == NOW + 60.0
        else:
            raise AssertionError('expected NoCredentialsAvailableError')
        # Clear for next steps.
        for i in range(3):
            clear_rate_limited('openai', i)

        # iter_active_credentials yields all 3.
        active = list(iter_active_credentials('openai', vault_path=path, now=NOW))
        assert len(active) == 3, len(active)

        # last_rotation persisted to disk.
        v = load_vault(path)
        assert 'openai' in v.last_rotation
        assert 0 <= v.last_rotation['openai'] < 3
        print('OK select_credential round-trip')
        print('OK cool-down skip')
        print('OK all-cooled-down raises')
        print('OK iter_active_credentials')
        print('OK last_rotation persisted to disk:', v.last_rotation)

    asyncio.run(run())
    " 2>&1 | tee /tmp/phase019-smoke.log

    # Verify SQLite-style mode check on the vault file (POSIX only)
    if [ "$(uname -s)" = "Linux" ] || [ "$(uname -s)" = "Darwin" ]; then
      MODE="$(stat -f '%Lp' "$SMOKE_VAULT" 2>/dev/null || stat -c '%a' "$SMOKE_VAULT")"
      [ "$MODE" = "600" ] || { echo "FAIL: vault mode is $MODE, expected 600 (Phase 012 chmod 0600 contract preserved)"; exit 1; }
    fi

    rm -rf "$SMOKE_DIR"
    ```

    Required outcome: every line above prints "OK ..."; no `FAIL:` line; exit code 0.
